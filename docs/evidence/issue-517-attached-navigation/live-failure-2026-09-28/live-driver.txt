function codexVersion(executable) {
  try {
    return execFileSync(executable, ["--version"], { encoding: "utf8" }).trim();
  } catch (error) {
    throw new Error(`Could not read a version from ${executable}: ${error?.message ?? error}`);
  }
}

// Standard managed Codex subscription lease; auth remains in its production home.
function providerExecution(profile) {
  const definition = { id: profile.providerId, adapterId: profile.adapterId, accessContract: profile.contract };
  return async () => ({
    definition,
    descriptor: { adapterId: definition.adapterId, accessContract: definition.accessContract,
      implementationVersion: productionProviderAdapterRegistry.get(definition.adapterId).implementationVersion },
    runtime: { async executionAccess() { return {
      kind: "managed-runtime", runtimeId: "codex", version: codexVersion(profile.codexExecutable), executable: profile.codexExecutable,
      environment: { CODEX_HOME: profile.codexHome, RELAYER_CODEX_BINARY: profile.codexExecutable,
        ...(process.env.PATH ? { PATH: process.env.PATH } : {}), ...(process.env.HOME ? { HOME: process.env.HOME } : {}) },
    }; } },
    async release() {},
  });
}

async function prepareRoute({ session, productServer, runtime, resolver, profile }) {
  const { providerId, modelId } = profile;
  const catalog = {
    providerId,
    label: providerId,
    connected: true,
    models: [{
      id: modelId,
      label: modelId,
      order: 0,
      visible: true,
      available: true,
      providerDefault: true,
      metadata: {},
    }],
    systemFamily: { key: providerId, name: providerId, modelIds: [modelId] },
  };
  await productServer.publishProviderCatalog(catalog);
  const readiness = createHarnessReadinessCoordinator({
    configurations: runtime.session.configurations,
    digestConfiguration: runtime.session.digestConfiguration,
    runtimeRequirements: {},
    prepareRecipe: async (recipeId) => productionHarnessRuntimeDescriptor(await resolver.prepare(recipeId)),
    checkers: { [profile.implementation]: async () => ({ available: true }) },
    publishAvailability: async (updates) => {
      await productServer.publishHarnessReadiness(updates);
      await runtime.recordHarnessReadiness(updates);
    },
  });
  const { readyHarnessIds, routeResults } = await readiness.evaluate({
    trigger: "connect",
    providerDefinition: { id: providerId, adapterId: profile.adapterId, accessContract: profile.contract },
    models: [{ id: modelId, visible: true, available: true }],
  });
  if (!readyHarnessIds.includes(profile.harness)) {
    const reason = routeResults.find(({ harnessId }) => harnessId === profile.harness)?.unavailableReason;
    throw new Error(`Harness ${profile.harness} is not ready for ${providerId}/${modelId}${reason ? `: ${reason.code}` : ""}.`);
  }

  const settings = await productRequest(session, "/api/model-settings");
  const existing = settings.families.find(family => family.name === "Live run models"
    && family.enabled && family.members.some(member => member.providerId === providerId && member.modelId === modelId));
  const family = existing ?? await productRequest(session, "/api/model-families", {
    method: "POST",
    body: JSON.stringify({
      name: "Live run models",
      enabled: true,
      members: [{ providerId, modelId }],
    }),
  });
  await productRequest(session, "/api/model-selection/validate", {
    method: "POST",
    body: JSON.stringify({ harnessId: profile.harness, familyId: family.id, providerId, modelId }),
  });
  return family;
}

async function fileIdentity(path) {
  const bytes = await readFile(path);
  return { path, byteLength: bytes.length, sha256: createHash("sha256").update(bytes).digest("hex") };
}
async function treeIdentity(directory) {
  const entries = [];
  async function visit(folder) {
    for (const entry of (await readdir(folder, { withFileTypes: true })).sort((a, b) => a.name.localeCompare(b.name))) {
      const path = join(folder, entry.name);
      if (entry.isDirectory()) await visit(path);
      else entries.push(await fileIdentity(path));
    }
  }
  await visit(directory);
  return { directory, entries, sha256: createHash("sha256").update(JSON.stringify(entries)).digest("hex") };
}

async function settle(threadId, interactionId) {
  const deadline = Date.now() + 15 * 60 * 1000;
  let previous;
  for (;;) {
    const detail = await productRequest(session, `/api/threads/${threadId}`);
    const interaction = detail.interactions.find(value => value.id === interactionId);
    if (!interaction) throw new Error("Missing live interaction projection");
    if (interaction.completionStatus !== previous) {
      previous = interaction.completionStatus;
      console.log(JSON.stringify({ interactionId, completionStatus: previous }));
    }
    if (interaction.completionStatus === "accepted") return { detail, interaction };
    if (interaction.latestAttempt?.finishedAt && interaction.latestAttempt.outcome !== "accepted") {
      await save(`failed-interaction-${interactionId}`, detail);
      throw new Error(`Live interaction ${interactionId} settled ${interaction.latestAttempt.outcome}; no retry performed`);
    }
    if (Date.now() > deadline || interaction.completionStatus === "waiting_for_approval") {
      await save(`blocked-interaction-${interactionId}`, detail);
      throw new Error(`Live interaction ${interactionId} requires attention (${interaction.completionStatus})`);
    }
    await new Promise(resolveWait => setTimeout(resolveWait, 1000));
  }
}
